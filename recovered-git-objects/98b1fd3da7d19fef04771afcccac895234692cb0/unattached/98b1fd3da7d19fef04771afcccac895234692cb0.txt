import { useEffect, useMemo, useRef, useState } from "react";
import { useNavigate } from "react-router-dom";
import { bulkUpsertProjectEnvVars, connectGithubAppInstallation, createProject, deployNormalV1, generatePreflightReport, getGithubConnectionStatus, getGithubRepositories, inspectGithubRepository, runStackDetection } from "../api/projectApi.js";
import ErrorState from "../components/common/ErrorState.jsx";
import LoadingState from "../components/common/LoadingState.jsx";
import { parseEnvPaste } from "../utils/envPaste.js";

export default function NewProject() {
  const navigate = useNavigate(); const inFlight = useRef(false);
  const [status, setStatus] = useState(null); const [repositories, setRepositories] = useState([]); const [repository, setRepository] = useState(""); const [branch, setBranch] = useState(""); const [branches, setBranches] = useState([]); const [envPaste, setEnvPaste] = useState(""); const [error, setError] = useState(""); const [busy, setBusy] = useState(true); const [savedKeys, setSavedKeys] = useState([]);
  const parsed = useMemo(() => parseEnvPaste(envPaste), [envPaste]);
  async function refresh() { const [connection, list] = await Promise.all([getGithubConnectionStatus(), getGithubRepositories()]); setStatus(connection); setRepositories(list.repositories || []); }
  useEffect(() => { const id = new URLSearchParams(location.search).get("installation_id"); (id ? connectGithubAppInstallation(id) : Promise.resolve()).then(refresh).catch((caught) => setError(caught.message)).finally(() => setBusy(false)); }, []);
  async function choose(value) { const item = repositories.find((entry) => entry.fullName === value); setRepository(value); setBranch(item?.defaultBranch || ""); setBranches([]); setError(""); if (!value) return; try { const details = (await inspectGithubRepository(value)).repository; setBranches(details.branches || []); setBranch(details.defaultBranch || details.branches?.[0] || ""); } catch (caught) { setError(caught.message); } }
  async function deploy() { if (inFlight.current || !repository || !branch || parsed.errors.length) return; inFlight.current = true; setBusy(true); setError(""); try { let project; try { project = (await createProject({ repositoryFullName: repository, targetBranch: branch, name: repository.split("/").pop() })).project; } catch (caught) { if (caught.code === "EXISTING_PROJECT" || caught.payload?.code === "EXISTING_PROJECT") project = caught.payload.existingProject; else throw caught; } await runStackDetection(project.id); if (parsed.entries.length) { await bulkUpsertProjectEnvVars(project.id, parsed.entries.map(({ key, value, isSecret }) => ({ key, value, isSecret, scope: "runtime" }))); setSavedKeys(parsed.entries.map(({ key }) => key)); } await generatePreflightReport(project.id); await deployNormalV1(project.id); navigate(`/projects/${project.id}`); } catch (caught) { setError(caught.message); } finally { inFlight.current = false; setBusy(false); } }
  if (busy && !status) return <LoadingState message="Checking GitHub App access…" />;
  return <div className="workspace-page new-project-page"><header className="workspace-heading"><div><p className="eyebrow">Deploy</p><h1>Deploy a GitHub repository</h1><p>Choose an authorized repository and branch. DeployGuard handles the analysis and cloud delivery.</p></div></header>{error ? <ErrorState message={error} /> : null}{!status?.connected ? <section className="state warning"><strong>Connect GitHub App</strong><p>{status?.message || "GitHub App access is required."}</p>{status?.availableInstallations?.map((item) => <button className="button" key={item.installationId} onClick={() => connectGithubAppInstallation(item.installationId).then(refresh).catch((caught) => setError(caught.message))} type="button">Connect {item.accountLogin}</button>)}{status?.installUrl ? <a className="secondary-button" href={status.installUrl}>Install GitHub App</a> : null}</section> : <section className="panel-flat"><label className="field"><span>Authorized repository</span><select onChange={(event) => void choose(event.target.value)} value={repository}><option value="">Select a repository</option>{repositories.map((item) => <option key={item.id || item.fullName} value={item.fullName}>{item.fullName}</option>)}</select></label><label className="field"><span>Branch</span><select disabled={!repository} onChange={(event) => setBranch(event.target.value)} value={branch}>{branches.map((item) => <option key={item} value={item}>{item}</option>)}</select></label><label className="field"><span>Optional application .env</span><textarea onChange={(event) => setEnvPaste(event.target.value)} placeholder={'# Optional\nAPI_URL=https://example.test'} rows="7" value={envPaste} /><small>Only application-owned values belong here. Leave empty when none are needed.</small></label>{parsed.errors.map((message) => <p className="state error" key={message}>{message}</p>)}{savedKeys.length ? <p className="state success">Saved keys: {savedKeys.join(", ")}</p> : null}<button className="button" disabled={busy || !repository || !branch || parsed.errors.length} onClick={() => void deploy()} type="button">{busy ? "Preparing deployment…" : "Deploy"}</button></section>}</div>;
}
