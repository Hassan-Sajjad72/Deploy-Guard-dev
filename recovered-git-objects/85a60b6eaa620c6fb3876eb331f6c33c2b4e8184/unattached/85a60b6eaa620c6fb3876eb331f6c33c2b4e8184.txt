  | "not_started"
  | "pending"

  | "passed"
  | "failed"
  | "blocked"
  | "warning"
  | "unavailable"

  stage: string;
  label: string;
  status: PipelineStageStatus;
  required: boolean;
  startedAt: Date | null;
  endedAt: Date | null;
  durationMs: number | null;
