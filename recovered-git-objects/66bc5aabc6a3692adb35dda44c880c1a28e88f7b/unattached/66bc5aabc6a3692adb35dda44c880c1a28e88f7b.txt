import { useEffect, useState } from "react";
import { Link } from "react-router-dom";
import { getUsers, updateUserRole } from "../api/adminApi.js";
import { getWorkspaceSummary } from "../api/projectApi.js";
import EmptyState from "../components/common/EmptyState.jsx";
import ErrorState from "../components/common/ErrorState.jsx";
import LoadingState from "../components/common/LoadingState.jsx";
import UserTable from "../components/admin/UserTable.jsx";
import { projectStatePresentation } from "../utils/projectStatePresentation.js";

export default function AdminUsers() {
  const [users, setUsers] = useState([]);
  const [error, setError] = useState("");
  const [success, setSuccess] = useState("");
  const [isLoading, setIsLoading] = useState(true);
  const [updatingUserId, setUpdatingUserId] = useState(null);
  const [projects, setProjects] = useState([]);

  async function loadUsers() {
    setError("");
    setIsLoading(true);

    try {
      const [response, workspace] = await Promise.all([getUsers(), getWorkspaceSummary()]);
      setUsers(response?.users || []);
      setProjects(workspace?.summaries || []);
    } catch (caughtError) {
      setError(caughtError.message);
    } finally {
      setIsLoading(false);
    }
  }

  useEffect(() => {
    loadUsers();
  }, []);

  async function handleRoleChange(userId, role) {
    setError("");
    setSuccess("");
    setUpdatingUserId(userId);

    try {
      const response = await updateUserRole(userId, role);
      const updatedUser = response?.user;

      setUsers((currentUsers) =>
        currentUsers.map((user) =>
          user.id === userId ? { ...user, ...updatedUser } : user
        )
      );
      setSuccess("User role updated.");
    } catch (caughtError) {
      setError(caughtError.message);
    } finally {
      setUpdatingUserId(null);
    }
  }

  return (
    <div className="grid">
      <div className="page-header">
        <div>
          <h1>Admin</h1>
          <p className="muted">Read-only project state visibility and RBAC role management.</p>
        </div>
      </div>

      {error ? <ErrorState message={error} /> : null}
      {success ? <div className="state success">{success}</div> : null}
      {isLoading ? <LoadingState message="Loading users..." /> : null}
      {!isLoading && !error && users.length === 0 ? (
        <EmptyState message="No users found." />
      ) : null}
      {!isLoading && users.length > 0 ? (
        <UserTable
          onRoleChange={handleRoleChange}
          updatingUserId={updatingUserId}
          users={users}
        />
      ) : null}
      {!isLoading ? <section className="panel-flat" data-admin-project-state-source="current-state">
        <p className="eyebrow">Projects and operations</p>
        <h2>Authoritative project states</h2>
        {!projects.length ? <EmptyState message="No projects are available." /> : <div className="active-run-list">{projects.map(({ project, currentState }) => {
          const state = projectStatePresentation(currentState);
          return <article className="active-run-item" data-authoritative-state={state.state} key={project.id}>
            <span className="active-run-identity"><strong>{project.name}</strong><small>{state.headline}</small></span>
            <strong>{state.state}</strong>
            <Link className="text-link" to={`/projects/${project.id}`}>Open project</Link>
          </article>;
        })}</div>}
      </section> : null}
    </div>
  );
}
